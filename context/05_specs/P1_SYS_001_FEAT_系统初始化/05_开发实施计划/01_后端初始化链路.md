```go
    Exists(ctx context.Context) (bool, error)
}

}

func NewSystemInitializationRepository(db *gorm.DB) SystemInitializationRepository {

```

**实现契约：**

`service/setup.go`：

```go
type SetupService interface {
    GetStatus(ctx context.Context) (*SetupStatus, error)
    Initialize(ctx context.Context, username, name, passwordCipher, keyID string) (*SetupResult, error)
}

type SetupStatus struct {
    Initialized        bool
    DBType             string
    DatabaseConnected  bool
    RedisConnected     bool
    JWTSecretSecure    bool
    BlockSubmit        bool
}

type SetupResult struct {
    Initialized bool
    Account     *AccountDTO  // 复用 account 域 AccountDTO
}

type setupService struct {
    db            *gorm.DB
    systemInitRepo repository.SystemInitializationRepository
    accountRepo   AccountRepository
    decryptor     PasswordDecryptor
    dbProbe       func(context.Context) bool
    redisProbe    func(context.Context) bool
    jwtSecure     bool
}

func NewSetupService(db *gorm.DB, systemInitRepo repository.SystemInitializationRepository, accountRepo AccountRepository, decryptor PasswordDecryptor, dbProbe, redisProbe func(context.Context) bool, jwtSecure bool) SetupService
```

**关键逻辑：**

GetStatus（specs §5.1.2）：
1. `initialized, _ := systemInitRepo.Exists(ctx)`，查询出错时 initialized 取 false 并 slog.Error
2. `databaseConnected := dbProbe(ctx)`、`redisConnected := redisProbe(ctx)`
3. `dbType := model.Current()`（model 包级函数返回 "sqlite"/"mysql"/"postgres"）
4. `blockSubmit := !databaseConnected || !redisConnected`（specs 规则3，jwtSecure 不参与阻断）
5. 组装 SetupStatus 返回

Initialize（specs §5.2.2）：
1. `exists, _ := systemInitRepo.Exists(ctx)`，exists 为 true 返 `NewError(errcode.SystemAlreadyInitialized)`（1101）
2. `if !dbProbe(ctx) || !redisProbe(ctx) { return NewError(errcode.EnvironmentNotReady) }`（1102）
3. `if !validateUsername(username) { return NewError(errcode.BadRequest) }`（1400，复用 account 同包函数）
4. `if utf8.RuneCountInString(name) > 20 || len(name) == 0 { return NewError(errcode.BadRequest) }`（1400）
5. `plain, err := decryptor.Decrypt(ctx, keyID, passwordCipher)`，err != nil 返 `NewError(errcode.BadRequest)`（1400）
6. `if !validatePassword(plain) { return NewError(errcode.PasswordInvalid) }`（1007，复用 account 同包函数）
7. `existing, _ := accountRepo.FindByUsername(ctx, username)`，existing != nil（非 ErrRecordNotFound）返 `NewError(errcode.UsernameExists)`（1005）
8. `err := db.Transaction(func(tx *gorm.DB) error {` 内部：
   - `hash, _ := bcrypt.GenerateFromPassword([]byte(plain), bcrypt.DefaultCost)`
   - `acc := &domain.Account{Username: username, Name: name, PasswordHash: string(hash), Enabled: true}`
   - `if e := tx.Create(acc).Error; e != nil { return e }`（雪花 ID 由 GORM 全局 Create 回调赋值）
   - `rec := &domain.SystemInitialization{CreatorAccountID: acc.ID, DBType: model.Current()}`
   - `return tx.Create(rec).Error`
   `})`，事务失败返 `NewError(errcode.Internal)`（1500）
9. 返回 `&SetupResult{Initialized: true, Account: toDTO(acc)}`（toDTO 复用 account 同包函数转 AccountDTO）

**依赖可解析：** import `context`、`crypto/...` 经 bcrypt、`gorm.io/gorm`、`sili-smart-hr/backend/internal/domain`、`sili-smart-hr/backend/internal/model`、`sili-smart-hr/backend/internal/pkg/errcode`、`sili-smart-hr/backend/internal/repository`、`golang.org/x/crypto/bcrypt`、`unicode/utf8`、`log/slog`。validateUsername/validatePassword/toDTO/AccountDTO/NewError 均为 service 同包既有（account.go 定义）。

**测试契约：** `service/setup_test.go`（package service_test），用真 SQLite 内存 + miniredis + 真 repo + fake decryptor + 可控 probe，复用 account_test.go 的 `fakeDecryptor`（同包）与 `wantCode` 辅助。每测试用 `gorm.Open(sqlite.Open(":memory:"))` + `AutoMigrate(&domain.Account{}, &domain.SystemInitialization{})` 建表，构造 `repository.NewAccountRepository(db)` 与 `repository.NewSystemInitializationRepository(db)` 与 `service.NewSetupService(db, sysRepo, accRepo, dec, dbProbe, redisProbe, true)`。

**核心断言：**
- `TestGetStatus_NotInitialized_NoBlock`：空库 + dbProbe/redisProbe 返 true + jwtSecure=true → `Initialized==false`、`DatabaseConnected==true`、`RedisConnected==true`、`BlockSubmit==false`、`DBType=="sqlite"`（SQLite 测试环境）
- `TestGetStatus_RedisDown_Block`：redisProbe 返 false → `BlockSubmit==true`、`RedisConnected==false`
- `TestInitialize_AlreadyInit_1101`：先成功 Initialize 一次，第二次返 `*service.Error{Code: 1101}`（用 `wantCode(t, err, 1101)`）
- `TestInitialize_DBDown_1102`：dbProbe 返 false → 返 `*service.Error{Code: 1102}`
- `TestInitialize_BadUsername_1400`：username="ab" → `Code: 1400`
- `TestInitialize_BadPassword_1007`：fakeDecryptor 返 "short" → `Code: 1007`
- `TestInitialize_DecryptFail_1400`：fakeDecryptor 返 err → `Code: 1400`
