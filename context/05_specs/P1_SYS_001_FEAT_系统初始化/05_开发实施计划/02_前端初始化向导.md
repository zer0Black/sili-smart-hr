```ts
import { httpClient } from '@/lib/http-client';
import type { InitializePayload, InitializeResult, SetupStatus } from '@/lib/contracts';

/** GET /api/setup/status：返回初始化状态与环境自检结果。 */
export async function fetchSetupStatus(): Promise<SetupStatus> {
  const { data } = await httpClient.get<SetupStatus>('/setup/status');
  return data;
}

/** POST /api/setup/initialize：创建首个账号并锁定初始化状态。 */
export async function initialize(payload: InitializePayload): Promise<InitializeResult> {
  const { data } = await httpClient.post<InitializeResult>('/setup/initialize', payload);
  return data;
}
```

`features/system/hooks.ts`（仿 account/hooks.ts）：
```ts
import { useMutation, useQuery } from '@tanstack/react-query';

import type { InitializePayload, InitializeResult, SetupStatus } from '@/lib/contracts';
import { queryClient } from '@/lib/query-client';

import { fetchSetupStatus, initialize } from './api';

/**
 * useSetupStatus：staleTime Infinity，首访探针在 __root beforeLoad 用 fetchQuery
 * 预取并缓存，向导页直接命中缓存，避免重复请求。
 */
export function useSetupStatus() {
  return useQuery<SetupStatus>({
    queryKey: ['setup', 'status'],
    queryFn: fetchSetupStatus,
    staleTime: Infinity,
  });
}

/** useInitialize：成功后废弃 setup status 缓存，下次导航 __root beforeLoad 重取为已初始化态。 */
export function useInitialize() {
  return useMutation<InitializeResult, Error, InitializePayload>({
    mutationFn: initialize,
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: ['setup', 'status'] });
    },
  });
}
```

**关键逻辑：** useSetupStatus 的 staleTime: Infinity 与全局 queryClient 的 30s staleTime 不同，因初始化状态单向锁定且首访探针需稳定缓存。useInitialize 的 onSuccess 只 invalidate 缓存，不做导航（导航与完成提示页过渡归组件控制）。

**验收锚点：**
- 核心断言：`pnpm type-check` 通过；api.ts 两个函数返回类型为 SetupStatus / InitializeResult
- 验证命令：`pnpm type-check`

**类型：** 代码任务（无测试，type-check 为检查门）

**文件：**
- 修改 `hr-frontend/src/routes/__root.tsx`

**specs 依据：**
- 章节范围：01_功能需求规格说明书.md §3.2（流程说明1 首访探针）、§4.1.1、§4.2.1；03_api_interface.md §3.1（首访探针消费）
- 业务规则索引：
  - [BR1] §4.1.4 规则1：探针据 system_initializations 记录存在性判定 initialized
  - [BR15] §3.2 流程说明1：未初始化时强制跳转向导页；接口失败（网络超时、5xx）按未初始化处理跳转向导页，由向导页自检复测暴露真实环境问题
  - [BR16] §4.1.1：已初始化后访问 /setup 重定向至登录页

**实现契约：**

`__root.tsx` 的 `Route` 增加 `beforeLoad`，在 `createRootRouteWithContext<RouterContext>()({...})` 配置对象中加入：

```ts
beforeLoad: async ({ context, location }) => {
  let initialized = false;
  try {
    const status = await context.queryClient.fetchQuery({
      queryKey: ['setup', 'status'],
      queryFn: fetchSetupStatus,
      staleTime: Infinity,
    });
    initialized = status.initialized;
  } catch {
    initialized = false; // 接口失败按未初始化处理（specs 3.2 流程说明1）
  }
  if (!initialized && location.pathname !== '/setup') {
    throw redirect({ to: '/setup' });
  }
  if (initialized && location.pathname === '/setup') {
    throw redirect({ to: '/login' });
  }
},
```

import 追加：`import { Outlet, createRootRouteWithContext, redirect } from '@tanstack/react-router';`（在现有 import 行加 redirect）、`import { fetchSetupStatus } from '@/features/system/api';`。

**关键逻辑：**
- `context.queryClient` 来自 RouterContext，beforeLoad 通过入参 `{context}` 取得
- fetchQuery + staleTime: Infinity：首次访问预取并长期缓存，后续导航命中缓存不再请求；初始化完成后 useInitialize 的 invalidate 触发下次导航重取
- 三条分支：未初始化且非 /setup → 跳 /setup；已初始化且在 /setup → 跳 /login；其余放行（含已初始化访问 /login、/_authenticated 等正常路径）
- 接口失败 catch：initialized 取 false，访问任何非 /setup 页都跳 /setup，向导页 useSetupStatus 复测时自检项全红暴露环境问题，避免探针失败直接放行卡在登录页
- beforeLoad 在 __root，先于 _authenticated/route.tsx 的 token 校验执行：已初始化 + token 有效才放行受保护页

**验收锚点：**
- 核心断言：`pnpm type-check` 通过
- 手动验证（后端起后）：未初始化态访问 `http://127.0.0.1:3000/login` 自动跳 `/setup`；初始化完成后访问 `/setup` 自动跳 `/login`
- 验证命令：`pnpm type-check`，`pnpm dev` 后浏览器验证重定向
